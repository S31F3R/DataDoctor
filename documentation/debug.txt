Imports System.IO
Imports System.Threading

Module Module1
    Private pythonProcess As Process = Nothing
    Sub Main()
        'Get the directory where the EXE is rsunning
        Dim exeDirectory As String = AppDomain.CurrentDomain.BaseDirectory
        Dim pythonScriptPath As String = Path.Combine(exeDirectory, "Project Files\DataDoctor.py")

        'Add handler for cleanup on .ext exit/crash
        AddHandler AppDomain.CurrentDomain.ProcessExit, AddressOf cleanupPythonProcess

        'Check if the Python script exists
        If Not File.Exists(pythonScriptPath) Then
            Console.WriteLine("Error: DataDoctor.py not found at {0}. Ensure DataDoctor folder is alongside this EXE.", pythonScriptPath)
            Console.WriteLine("Press any key to exit...")
            Console.ReadKey()
            Return
        End If

        'Launch the Python script
        Try
            Dim startInfo As New ProcessStartInfo With {
                .FileName = "python.exe",
                .Arguments = """" + pythonScriptPath + """", 'Quote path for spaces
                .WorkingDirectory = exeDirectory,
                .UseShellExecute = False,
                .RedirectStandardOutput = True,
                .RedirectStandardError = True,
                .CreateNoWindow = True
            }

            pythonProcess = New Process With {.StartInfo = startInfo}
            pythonProcess.EnableRaisingEvents = True
            pythonProcess.Start()

            'Wait for exit and reat output
            pythonProcess.WaitForExit()

            Dim pollTimeout As Integer = 1000
            Dim pollInterval As Integer = 100
            Dim polls As Integer = pollTimeout / pollInterval

            For i As Integer = 1 To polls
                If pythonProcess IsNot Nothing AndAlso Not pythonProcess.HasExited Then
                    cleanupPythonProcess(Nothing, Nothing)
                    Exit For
                End If

                Thread.Sleep(pollInterval)
            Next
        Catch ex As Exception
            Console.WriteLine("Press any key to exit...")
            Console.ReadKey()
        End Try
    End Sub

    Private Sub cleanupPythonProcess(sender As Object, e As EventArgs)
        If pythonProcess IsNot Nothing AndAlso Not pythonProcess.HasExited Then
            'Graceful close first
            pythonProcess.CloseMainWindow()
            Thread.Sleep(2000) 'Wait 2 seconds

            'Hard kill if still running
            If Not pythonProcess.HasExited Then
                pythonProcess.Kill()
                pythonProcess.WaitForExit(5000) 'Wait up to 5 seconds for kill effect
            End If
        End If

        pythonProcess = Nothing
    End Sub
End Module
